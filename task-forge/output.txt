from pathlib import Path

# ==========================
# USER INPUT
# ==========================

SOURCE_FOLDER = input("Enter the source folder path: ").strip()
OUTPUT_FILE = input("Enter the output file path (e.g., C:\\combined.txt): ").strip()

source = Path(SOURCE_FOLDER).resolve()
output_path = Path(OUTPUT_FILE).resolve()

# ==========================
# FILE TYPES TO INCLUDE
# ==========================

EXTENSIONS = {
    ".txt",
    ".md",
    ".py",
    ".java",
    ".kt",
    ".kts",
    ".js",
    ".jsx",
    ".ts",
    ".tsx",
    ".html",
    ".css",
    ".scss",
    ".sass",
    ".json",
    ".xml",
    ".yaml",
    ".yml",
    ".properties",
    ".gradle",
    ".sql",
    ".c",
    ".cpp",
    ".cc",
    ".h",
    ".hpp",
    ".cs",
    ".go",
    ".rs",
    ".php",
    ".rb",
    ".swift",
    ".dart",
    ".sh",
    ".bat",
    ".ps1",
    ".env",
    ".gitignore"
}

# ==========================
# FOLDERS TO IGNORE
# ==========================

IGNORE_DIRS = {
    # Git
    ".git",

    # Node.js
    "node_modules",
    ".next",
    ".npm",
    ".pnpm-store",
    ".turbo",

    # React / Vite
    "dist",
    "build",
    "coverage",

    # Python
    "__pycache__",
    ".venv",
    "venv",
    "env",
    ".pytest_cache",
    ".mypy_cache",

    # Java / Spring Boot
    "target",
    ".gradle",
    "out",
    "bin",

    # Maven wrapper
    ".mvn",

    # IDEs
    ".idea",
    ".vscode",

    # OS
    ".DS_Store"
}

print("\nScanning files...\n")

files_processed = 0
files_skipped = 0

with open(output_path, "w", encoding="utf-8") as outfile:

    for file in sorted(source.rglob("*")):

        # Skip ignored folders
        if any(part in IGNORE_DIRS for part in file.parts):
            continue

        # Skip output file
        if file.resolve() == output_path:
            continue

        # Skip non-files
        if not file.is_file():
            continue

        # Skip unwanted extensions
        if file.suffix.lower() not in EXTENSIONS:
            continue

        print(f"Reading: {file.relative_to(source)}")

        outfile.write("=" * 100 + "\n")
        outfile.write(f"FILE: {file.relative_to(source)}\n")
        outfile.write("=" * 100 + "\n\n")

        try:
            with open(file, "r", encoding="utf-8") as infile:
                outfile.write(infile.read())
                files_processed += 1

        except Exception as e:
            outfile.write(f"[Could not read file: {e}]")
            files_skipped += 1

        outfile.write("\n\n")

print("\n" + "=" * 50)
print("Completed!")
print("=" * 50)
print(f"Files merged   : {files_processed}")
print(f"Files skipped  : {files_skipped}")
print(f"Output saved to: {output_path}")

### Maven Parent overrides

Due to Maven's design, elements are inherited from the parent POM to the project POM.
While most of the inheritance is fine, it also inherits unwanted elements like `<license>` and `<developers>` from the parent.
To prevent this, the project POM contains empty overrides for these elements.
If you manually switch to a different parent and actually want the inheritance, you need to remove those overrides.

package com.sudhanva.task_forge.common.exception;

import java.time.Instant;
import java.time.LocalDateTime;

import org.springframework.http.HttpStatus;
import org.springframework.http.ResponseEntity;
import org.springframework.http.converter.HttpMessageNotReadableException;
import org.springframework.web.bind.MethodArgumentNotValidException;
import org.springframework.web.bind.annotation.ExceptionHandler;
import org.springframework.web.bind.annotation.RestControllerAdvice;
import org.springframework.web.method.annotation.MethodArgumentTypeMismatchException;

import com.sudhanva.task_forge.common.api.ErrorCode;
import com.sudhanva.task_forge.common.api.ErrorResponseDto;

@RestControllerAdvice
public class GlobalExceptionHandler {
    
    @ExceptionHandler(Exception.class)
    public ResponseEntity<ErrorResponseDto> globalHandler(
        Exception exception
    ){
        ErrorResponseDto response = ErrorResponseDto.builder()
            .success(false)
            .errorCode(ErrorCode.INTERNAL_SERVER_ERROR)
            .message(exception.getMessage())
            .timestamp(Instant.now())
            .build();

        return ResponseEntity.status(HttpStatus.INTERNAL_SERVER_ERROR).body(response);
    }

    @ExceptionHandler(HttpMessageNotReadableException.class)
    public ResponseEntity<ErrorResponseDto> handleInvalidJson(
        HttpMessageNotReadableException exception
    ) {

        ErrorResponseDto response = ErrorResponseDto.builder()
            .success(false)
            .errorCode(ErrorCode.BAD_REQUEST)
            .message(exception.getMessage())
            .timestamp(Instant.now())
            .build();
        return ResponseEntity.badRequest().body(response);
    }
    
    @ExceptionHandler(MethodArgumentNotValidException.class)
    public ResponseEntity<ErrorResponseDto> handleValidation(
        MethodArgumentNotValidException exception
    ) {

        ErrorResponseDto response = ErrorResponseDto.builder()
            .success(false)
            .errorCode(ErrorCode.BAD_REQUEST)
            .message(exception.getMessage())
            .timestamp(Instant.now())
            .build();
        return ResponseEntity.badRequest().body(response);
    }

            .errorCode(ErrorCode.BAD_REQUEST)
            .message(exception.getMessage())
            .timestamp(Instant.now())
            .build();

    }

        ThreadPoolTaskExecutor executor = new ThreadPoolTaskExecutor();

        executor.setCorePoolSize(2);
        executor.setMaxPoolSize(4);
        executor.setQueueCapacity(10);

        executor.initialize();
        return executor;
    }

package com.sudhanva.task_forge.executor;

import java.util.concurrent.ArrayBlockingQueue;
import java.util.concurrent.LinkedBlockingQueue;
import java.util.concurrent.ThreadFactory;
import java.util.concurrent.ThreadPoolExecutor;
import java.util.concurrent.TimeUnit;
import java.util.concurrent.atomic.AtomicInteger;

import org.springframework.stereotype.Component;

// Create a configured ThreadPoolExecutor 
// and return it to whoever requests it
// Custom ThreadPoolFactory for my Wants
@Component
public class JobExecutorFactory {

    // Thread Factory
    private final AtomicInteger counter = new AtomicInteger(1);

    private final ThreadFactory factory = 
            runnable -> new Thread(
                runnable,
                "job-worker-"+counter.getAndIncrement()
            );


    // Create and return the pool
    public ThreadPoolExecutor createExecutor(){
        ThreadPoolExecutor pool = new ThreadPoolExecutor(
            2, 
            4,
            // This is for new worker cretaed and how much time keep them alive if idle
            30, 
            TimeUnit.SECONDS,
            // Waits in push and pop operation when ful lor empty
            new ArrayBlockingQueue<>(10),
            // new LinkedBlockingQueue<>(),
            factory,
            new ThreadPoolExecutor.AbortPolicy()
        );

        return pool;
    }
   
}

package com.sudhanva.task_forge.job.api.controller;

import java.net.URI;
import java.util.UUID;

import org.springframework.http.HttpHeaders;
import org.springframework.http.ResponseEntity;
import org.springframework.web.bind.annotation.GetMapping;
import org.springframework.web.bind.annotation.PathVariable;
import org.springframework.web.bind.annotation.PostMapping;
import org.springframework.web.bind.annotation.RequestBody;
import org.springframework.web.bind.annotation.RequestMapping;
import org.springframework.web.bind.annotation.RestController;
import org.springframework.web.servlet.support.ServletUriComponentsBuilder;

import com.sudhanva.task_forge.common.api.ApiResponse;
import com.sudhanva.task_forge.job.api.dto.JobResponse;
import com.sudhanva.task_forge.job.api.dto.SubmitJobRequest;
import com.sudhanva.task_forge.job.model.Job;
import com.sudhanva.task_forge.job.service.JobService;

import jakarta.validation.Valid;
import lombok.RequiredArgsConstructor;

@RestController
@RequestMapping("/api/jobs")
@RequiredArgsConstructor
public class JobController {

    private final JobService jobService;


    // Utility Methods

    private JobResponse mapToJobResponse(
        Job job
    ){
        JobResponse data = new JobResponse(job.id(), job.status(),job.type());
        return data;
    }
    

    // Mappings
    @PostMapping("/post-job")
    public ResponseEntity<ApiResponse<JobResponse>> acceptJob(
        @Valid @RequestBody SubmitJobRequest request
    ){

        Job job = jobService.submit(request);

        // Location URI
        URI location = ServletUriComponentsBuilder
                            .fromCurrentContextPath()
                            .path("/api/jobs/{id}")
                            .buildAndExpand(job.id())
                            .toUri();
        
        // map to response
        ApiResponse<JobResponse> data = new ApiResponse<>(
            true,
            "Job Accepted: "+job.id(),
            mapToJobResponse(job)
        );

        return ResponseEntity
                    .accepted()
                    .location(location)
                    .body(data);
    }


    @GetMapping("/{id}")
    public ResponseEntity<ApiResponse<JobResponse>> get(@PathVariable UUID id) {
        Job job = jobService.getJob(id);

        // map to response
        ApiResponse<JobResponse> data = new ApiResponse<>(
            true,
            "Job Status",
            mapToJobResponse(job)
        );
        return ResponseEntity.ok(data);
    }

}

        return ResponseEntity.status(HttpStatus.CONFLICT).body(response);

        return ResponseEntity.status(HttpStatus.NOT_FOUND).body(response);

package com.sudhanva.task_forge.job.interfaces;

// Interface for each handlers
// P specifc payload class
// R result retun by the handler
public interface JobHandler <P extends JobPayload, R>{
    
    // gives what type of payload is
    JobType type(); 
    // process the types payload and return a result
    R execute(P payload) throws Exception;
    // To retorve the class 
    // beacuse while casting The regestry losses the class Information
    // P tells the compiler the type; Class<P> lets us check and cast objects to that type at runtime.
    Class<P> payloadType();

    default R executeUntyped(JobPayload payload) throws Exception {
        return execute(payloadType().cast(payload));
    }
}

package com.sudhanva.task_forge.job.interfaces;

import java.util.List;
import java.util.Map;
import java.util.stream.Collectors;

import org.springframework.stereotype.Component;

// dleagets teh work to appropriate handler
@Component
public class JobHandlerRegistry {
    
    private final Map<JobType,JobHandler<?,?>> jobHandlers;

    public JobHandlerRegistry(List<JobHandler<?,?>> handlers){
        jobHandlers =  handlers.stream().collect(
            Collectors.toUnmodifiableMap( 
                handler -> handler.type(), 
                handler -> handler
            )
        );
    }

    public JobHandler<?,?> getHandler(JobType type){
        JobHandler<?,?> handler = jobHandlers.get(type);
        if (handler == null){
           throw new IllegalArgumentException(
                "No handler for: " + type +
                ". Registered handlers: " + jobHandlers.keySet()
            );
        }
            
        return handler;
    }

    // The compiler still cannot prove that the handler and payload match
    // The runtime class check provides that guarantee
    
}

import java.util.Set;
import java.util.UUID;
import java.util.concurrent.RejectedExecutionException;
// import java.util.concurrent.ThreadPoolExecutor;
import java.util.stream.Collectors;

import org.slf4j.Logger;
import org.slf4j.LoggerFactory;
import org.springframework.scheduling.concurrent.ThreadPoolTaskExecutor;
import org.springframework.stereotype.Service;

import tools.jackson.databind.JsonNode;
import tools.jackson.databind.ObjectMapper;

import com.sudhanva.task_forge.job.api.dto.SubmitJobRequest;
import com.sudhanva.task_forge.job.exceptions.ExecutorShutdownException;
import com.sudhanva.task_forge.job.exceptions.InvalidPayloadException;
import com.sudhanva.task_forge.job.exceptions.JobNotFoundException;
import com.sudhanva.task_forge.job.exceptions.QueueFullException;
import com.sudhanva.task_forge.job.interfaces.JobHandler;
import com.sudhanva.task_forge.job.interfaces.JobHandlerRegistry;
import com.sudhanva.task_forge.job.interfaces.JobPayload;
import com.sudhanva.task_forge.job.model.Job;
import com.sudhanva.task_forge.job.model.enums.JobStatus;
import com.sudhanva.task_forge.job.store.JobStore;

import jakarta.validation.ConstraintViolation;
import jakarta.validation.Validator;
import lombok.RequiredArgsConstructor;

// Test These
// good:           {"type":"ECHO","payload":{"message":"hi"}}   -> 202
// wrong shape:    {"type":"ECHO","payload":{"x":1}}            -> 400 (not 500)
// unknown type:   {"type":"NOPE","payload":{}}                 -> 400
// missing payload:{"type":"ECHO"}                              -> 400

@Service
@RequiredArgsConstructor
public class JobService {

    private final JobStore store;
    private final ThreadPoolTaskExecutor jobExecutor;
    private final JobHandlerRegistry registry;
    
    private final ObjectMapper objectMapper;
    private final Validator validator;

    private static final Logger log = LoggerFactory.getLogger(JobService.class);

    public Job submit(SubmitJobRequest request) {
        // checks that can fail come first
        // state changes last
        JobHandler<?, ?> handler = registry.getHandler(request.type());

        // convrting request payload to proper format
        JobPayload payload = toPayload(request.payload(), handler);

        Job job = new Job(UUID.randomUUID(), JobStatus.QUEUED, request.type());

        store.save(job);

        try {
            jobExecutor.execute(() -> runJob(job.id(), handler, payload));
        } catch (RejectedExecutionException e) {
            store.remove(job.id());
            throw jobExecutor.getThreadPoolExecutor().isShutdown()
                    ? new ExecutorShutdownException()
                    : new QueueFullException();
        } 
        return job;
    }

    // JSON -> payload object, then check its validation rules
    private JobPayload toPayload(JsonNode json, JobHandler<?, ?> handler) {
        JobPayload payload;

        try {
            payload = objectMapper.treeToValue(json, handler.payloadType());
        } catch (Exception e) {
            throw new InvalidPayloadException("Payload does not match the job type", e);
        }
        if (payload == null) {
            throw new InvalidPayloadException("Payload is empty");
        }

        // ? Not Understood
        Set<ConstraintViolation<JobPayload>> errors = validator.validate(payload);
        if (!errors.isEmpty()) {
            String message = errors.stream()
                    .map(v -> v.getPropertyPath() + " " + v.getMessage())
                    .collect(Collectors.joining(", "));
            throw new InvalidPayloadException(message);
        }

        return payload;
    }

    // runs on the worker thread
    private void runJob(UUID id, JobHandler<?, ?> handler, JobPayload payload) {
        if (!setStatus(id, JobStatus.RUNNING)) {
            // job was removed, nothing to run
            return;
        }
        try {
            System.out.println(Thread.currentThread().getName());
            handler.executeUntyped(payload);
            setStatus(id, JobStatus.SUCCEEDED);
        } catch (Exception e) {
            log.error("Job {} failed", id, e);
            setStatus(id, JobStatus.CANCELLED);
        } catch (Throwable t) {
            // This is for Error crash
            // the ThreaPoolExecutor later 
            // will replaces the dyied worker
            // thats why throwing the error again to crash the worker thread
            log.error("Job {} failed", id, t);
            setStatus(id, JobStatus.CANCELLED);
            if (t instanceof Error) {
                throw t;
            }
        }
    }

    // one place for "update + log if the job is missing"
    private boolean setStatus(UUID id, JobStatus status) {
        boolean updated = store.updateStatus(id, status);
        if (!updated) {
            log.warn("Job {} not found when setting {}", id, status);
        }
        return updated;
    }

    public Job getJob(UUID id) {
        Job job = store.findById(id).orElseThrow(
                () -> new JobNotFoundException(id));
        return job;
    }

}

package com.sudhanva.task_forge.job.store;

import java.util.Map;
import java.util.Optional;
import java.util.UUID;
import java.util.concurrent.ConcurrentHashMap;

import org.springframework.stereotype.Component;

import com.sudhanva.task_forge.job.exceptions.DuplicateJobException;
import com.sudhanva.task_forge.job.model.Job;
import com.sudhanva.task_forge.job.model.enums.JobStatus;

@Component
public class InMemoryJobStore implements JobStore{

    private final Map<UUID,Job> jobs = new ConcurrentHashMap<>();

    @Override
    public Optional<Job> findById(UUID id) {
        Job job = jobs.get(id);
        return Optional.ofNullable(job);
    }

    @Override
    public boolean remove(UUID id) {
        Job job = jobs.remove(id);
        return job != null;
    }

    @Override
    public void save(Job job) {
        // Service in the request stores the job so exception is handeled in global controller advice
        Job existing = jobs.putIfAbsent(job.id(), job);
         if (existing != null) {
            // thrown 409 CONLICT 
            throw new DuplicateJobException(job.id());
        }
    }

    @Override
    public boolean updateStatus(UUID id, JobStatus newStatus) {
        // update is done by worker thread so .orElseThrow() 
        // there is no http request waiting for this it wont reach controller advice
        // if u do normally like find and replce with new 
        // it will introduce race condition
        Job updated = jobs.computeIfPresent(
            id,
            // Takes the lambda's return value 
            // and replaces the existing value in the map.
            (key,oldJob) -> new Job(
                oldJob.id(),
                newStatus,
                oldJob.type()
            )
        );
        return updated != null;
    }
    
}

public class A2PoolDemo {
 
    public static void main(String[] args) {
        
        // JobExecutorFactory factory = new JobExecutorFactory();
        
        ThreadPoolTaskExecutor executor = new ThreadPoolTaskExecutor();

        for (int i = 1; i <= 16; i++) {

            int taskNo = i;

            try {
                executor.execute(() -> {
                    // Print task number and thread name
                    System.out.println("Task No: "+taskNo + " Thread Name: "+Thread.currentThread().getName());
                    try {
                        // Sleep for 30 seconds
                        // another thread can interrupt a sleeping thread before those 30 seconds are over.
                        Thread.sleep(9000);    
                    } catch (InterruptedException e) {
                        System.out.println("Worker interrupted"+Thread.currentThread().getName());
                        // pool.shutdown();
                    }
                });
                System.out.println("Job Accepted: "+taskNo);
            } catch (RejectedExecutionException e) {
                System.out.println("Job Rejected: "+taskNo);
            }
        }

        // Let running tasks finish, accept no new ones
        executor.shutdown();
    }
    
}

    @Test
    void manyThreads_savingDifferentJobs_allAreStored() throws Exception {

        List<Job> allJobs = new ArrayList<>();
        List<Future<?>> futures = new ArrayList<>();

    @Test
    void manyThreads_savingTheSameId_exactlyOneWins() throws Exception {
        Job job = newJob();
        ExecutorService pool = Executors.newFixedThreadPool(THREADS);
        CountDownLatch startGate = new CountDownLatch(1);
        AtomicInteger successes = new AtomicInteger();
        AtomicInteger duplicates = new AtomicInteger();
        List<Future<?>> futures = new ArrayList<>();

        for (int t = 0; t < THREADS; t++) {
            futures.add(pool.submit(() -> {
                startGate.await();
                try {
                    store.save(job);
                    successes.incrementAndGet();
                } catch (DuplicateJobException e) {
                    duplicates.incrementAndGet();
                }
                return null;
            }));
        }

        startGate.countDown();
        waitForAll(futures);
        pool.shutdown();

        assertEquals(1, successes.get());
        assertEquals(THREADS - 1, duplicates.get());

    @Test
    void manyThreads_updatingTheSameJob_noUpdateIsLostOrBroken() throws Exception {
        // each thread writes its own status many times.
        // At the end the job must still exist, with a valid status, and the same id and type.
        Job job = newJob();
        store.save(job);

        JobStatus[] statuses = {JobStatus.RUNNING, JobStatus.SUCCEEDED, JobStatus.CANCELLED};
        ExecutorService pool = Executors.newFixedThreadPool(THREADS);
        CountDownLatch startGate = new CountDownLatch(1);
        AtomicInteger falseResults = new AtomicInteger();
        List<Future<?>> futures = new ArrayList<>();

        for (int t = 0; t < THREADS; t++) {
            JobStatus mine = statuses[t % statuses.length];
            futures.add(pool.submit(() -> {
                startGate.await();
                for (int i = 0; i < 500; i++) {
                    if (!store.updateStatus(job.id(), mine)) {
                        falseResults.incrementAndGet();
                    }
                }
                return null;
            }));
        }

        startGate.countDown();
        waitForAll(futures);
        pool.shutdown();

        assertEquals(0, falseResults.get(), "job exists, so every update must return true");
        Job after = store.findById(job.id()).orElseThrow();
        assertEquals(job.id(), after.id());
        assertEquals(job.type(), after.type());
        assertTrue(List.of(statuses).contains(after.status()));
    }

    @Test
    void manyThreads_updateRacingWithRemove_neverLeavesAGhost() throws Exception {
        // One thread removes the job. Other threads keep updating it.
        // After the remove, the job must be gone for good. An update must not bring it back.
        int rounds = 300;

        for (int r = 0; r < rounds; r++) {
            InMemoryJobStore freshStore = new InMemoryJobStore();
            Job job = newJob();
            freshStore.save(job);

            ExecutorService pool = Executors.newFixedThreadPool(4);
            CountDownLatch startGate = new CountDownLatch(1);
            List<Future<?>> futures = new ArrayList<>();

            futures.add(pool.submit(() -> {
                startGate.await();
                freshStore.remove(job.id());
                return null;
            }));
            for (int t = 0; t < 3; t++) {
                futures.add(pool.submit(() -> {
                    startGate.await();
                    for (int i = 0; i < 50; i++) {
                        freshStore.updateStatus(job.id(), JobStatus.RUNNING);
                    }
                    return null;
                }));
            }

            startGate.countDown();
            waitForAll(futures);
            pool.shutdown();

            assertTrue(freshStore.findById(job.id()).isEmpty(), "ghost entry in round " + r);
        }
    }

    // waits for every task and rethrows the first error, so a failure inside a thread fails the test
    private void waitForAll(List<Future<?>> futures) throws Exception {
        for (Future<?> f : futures) {
            f.get(10, TimeUnit.SECONDS);
        }
    }
}

class ExecutorConfigTest {

    private ThreadPoolTaskExecutor executor;
    private final CountDownLatch gate = new CountDownLatch(1);

    @BeforeEach
    void setUp() {
        executor = new ExecutorConfig().jobExecutor();
        executor.initialize();     // Spring does this for a real bean
    }

    @AfterEach
    void tearDown() {
        gate.countDown();          // never leave workers stuck
        executor.getThreadPoolExecutor().shutdownNow();
    }

    @Test
    void settings_matchTheA2Pool() {
        assertEquals(2, executor.getCorePoolSize());
        assertEquals(4, executor.getMaxPoolSize());
        assertEquals(10, executor.getQueueCapacity());
    }

    @Test
    void workerThreads_useTheJobWorkerPrefix() throws Exception {
        AtomicReference<String> name = new AtomicReference<>();
        CountDownLatch done = new CountDownLatch(1);

        executor.execute(() -> {
            name.set(Thread.currentThread().getName());
            done.countDown();
        });

        assertTrue(done.await(5, TimeUnit.SECONDS));
        assertTrue(name.get().startsWith("job-worker-"));
    }

    @Test
    void sixteenBlockedJobs_fourteenAccepted_twoRejected() {
        int accepted = 0;
        int rejected = 0;

        for (int i = 0; i < 16; i++) {
            try {
                executor.execute(() -> {
                    try { gate.await(); } catch (InterruptedException ignored) { }
                });
                accepted++;
            } catch (RejectedExecutionException e) {
                rejected++;
            }
        }

        assertEquals(14, accepted);   // 2 core + 10 queue + 2 extra workers
        assertEquals(2, rejected);
    }

    @Test
    void shutdown_waitsForRunningAndQueuedJobs() throws Exception {
        AtomicInteger finished = new AtomicInteger();
        CountDownLatch firstStarted = new CountDownLatch(1);

        // 3 jobs: 2 run on core threads, 1 waits in the queue
        for (int i = 0; i < 3; i++) {
            executor.execute(() -> {
                firstStarted.countDown();
                try {
                    Thread.sleep(300);
                    finished.incrementAndGet();
                } catch (InterruptedException ignored) { }
            });
        }
        assertTrue(firstStarted.await(5, TimeUnit.SECONDS));

        executor.shutdown();          // should block until all 3 are done

        assertEquals(3, finished.get());
    }

    // ==========================================================
    // Fake handler: fully controlled by the tests
    // ==========================================================

    static class FakeHandler
            implements JobHandler<EchoPayload, String> {

        static final String FAIL_MESSAGE = "bad";

        final AtomicInteger calls = new AtomicInteger();

        final List<EchoPayload> received =
                new CopyOnWriteArrayList<>();

        final CountDownLatch firstStarted =
                new CountDownLatch(1);

        // Open by default. Tests can replace this with a blocking latch.
        volatile CountDownLatch gate =
                new CountDownLatch(0);

        @Override
        public String execute(EchoPayload payload) throws Exception {

            calls.incrementAndGet();

            received.add(payload);

            firstStarted.countDown();

            // Wait until the test opens the gate.
            gate.await();

            if (FAIL_MESSAGE.equals(payload.message())) {
                throw new IllegalStateException("boom");
            }

            return "ok";
        }

        @Override
        public JobType type() {
            return JobType.ECHO;
        }

        @Override
        public Class<EchoPayload> payloadType() {
            return EchoPayload.class;
        }
    }
